func TestReviewSimilarityNestedFunctionSignatureMustBlockSubstitution(t *testing.T) {
 h,e:=canonicalFixture(t,map[string]int{"cf-kilo":503,"cf-other":503},true)
 // This exact nested signature location is supported by the pinned Gemini
 // translator's openAIToolCallGeminiThoughtSignature, not an invented field.
 body:=[]byte(`{"messages":[{"role":"assistant","tool_calls":[{"id":"call_fixture","type":"function","function":{"name":"read_file","arguments":"{}","extra_content":{"google":{"thought_signature":"fixture-signature"}}}}]},{"role":"tool","tool_call_id":"call_fixture","content":"ok"}],"max_tokens":20}`)
 r,err:=canonicalRequestRequirements(body);if err!=nil {t.Fatal(err)}
 _,_,failure:=h.ExecuteWithAuthManager(context.Background(),"openai","glm-5.3-flash:free",body,"")
 substituted:=false;for _,call:=range e.calls {if strings.HasPrefix(call,"cf-alt/"){substituted=true}}
 if !r.OpaqueState || substituted {t.Fatalf("native-supported opaque signature escaped: OpaqueState=%v alternative_dispatched=%v calls=%v final_error=%v",r.OpaqueState,substituted,e.calls,failure)}
}
func TestReviewSimilarityExactReasoningModeMustBeSupported(t *testing.T) {
 r,err:=canonicalRequestRequirements([]byte(`{"reasoning_effort":"high","max_tokens":20}`));if err!=nil {t.Fatal(err)}
 m:=similarityModel("fixture-low-only")
 m.Thinking=&registry.ThinkingSupport{Levels:[]string{"low"}}
 // Prove that native thinking validation rejects this explicit mode.
 _,validation:=thinking.ValidateConfig(thinking.ThinkingConfig{Mode:thinking.ModeLevel,Level:thinking.ThinkingLevel("high")},m,"openai","openai",false)
 admitted:=canonicalCompatible(m,r)
 if admitted && validation!=nil {t.Fatalf("fallback admitted mode that native validator rejects: admitted=%v supported=%v error=%v",admitted,m.Thinking.Levels,validation)}
}
func TestReviewSimilarityToolSchemaIsNotMultimodalInput(t *testing.T) {
 // fileData is a legal user-defined tool argument, not an attached Gemini file.
 body:=[]byte(`{"messages":[{"role":"user","content":"list the files"}],"tools":[{"type":"function","function":{"name":"list_files","parameters":{"type":"object","properties":{"fileData":{"type":"string"}}}}}],"max_tokens":20}`)
 r,err:=canonicalRequestRequirements(body);if err!=nil {t.Fatal(err)}
 good:=similarityModel("fixture-text-tools")
 if r.Modalities["file"] || !canonicalCompatible(good,r) {t.Fatalf("tool schema misclassified as media: modalities=%v compatible=%v",r.Modalities,canonicalCompatible(good,r))}
}
func TestReviewSimilarityPublisherNamespacesMustNotProveContinuity(t *testing.T) {
 a:="publisher-a/agent-5-coder";b:="publisher-b/agent-5-coder"
 reference:=similarityModel(a);candidate:=similarityModel(b)
 r:=canonicalRequirements{Continuity:true}
 got:=canonicalScore(a,reference,candidate,r,nil)
 if got.Continuity {t.Fatalf("unverified publishers collapsed into same family: requested=%s candidate=%s continuity=%v score=%v",a,b,got.Continuity,got.Components)}
}
